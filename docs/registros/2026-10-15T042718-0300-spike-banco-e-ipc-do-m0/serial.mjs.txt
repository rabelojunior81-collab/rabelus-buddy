import { fork } from 'node:child_process';
import net from 'node:net';
import { mkdtempSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
const dir = mkdtempSync(join(tmpdir(), 'spike-m0s-'));
const ep = process.platform === 'win32' ? String.raw`\\.\pipe\rbuddy-serial-` + process.pid : join(dir, 's.sock');
const here = new URL('.', import.meta.url).pathname.replace(/^\/([A-Za-z]:)/, '$1');
const p = await new Promise((r) => { const c = fork(join(here, 'dono.mjs'), [join(dir, 'n.db'), ep]); c.on('message', () => r(c)); });
const s = net.connect(ep); s.setEncoding('utf8'); await new Promise((r) => s.on('connect', r));
const lat = []; let buf = '';
for (let k = 0; k < 2000; k++) {
  const t = performance.now();
  await new Promise((r) => { const on = (d) => { buf += d; if (buf.includes('\n')) { buf = ''; s.off('data', on); r(); } }; s.on('data', on); s.write(JSON.stringify({ id: k, session: 'x', kind: 'passo', body: 'z' }) + '\n'); });
  lat.push(performance.now() - t);
}
const q = (x) => [...lat].sort((a, b) => a - b)[Math.floor((lat.length - 1) * x)].toFixed(3);
console.log(JSON.stringify({ teste: 'latência ida e volta, 1 pedido por vez', n: lat.length, p50ms: q(0.5), p95ms: q(0.95), p99ms: q(0.99) }));
s.end(); p.kill('SIGKILL');
